\startenvironment *

% DHR-Override (Demo): andere Schrift als das JNDF-Baseline (jds).
% Journal-Level-Shadowing: diese Datei ersetzt pub/_assets/context/_layout_doc_fonts.tex
% fuer alle Artikel unter dhr/. Roman: TeX Gyre Pagella (statt Cardo),
% Sans: TeX Gyre Heros (statt Myriad Pro) — beides ConTeXt-gebindet.

% BIDI
\setupdirections[bidi=global,method=unicode]

% Kursivkorrektur
\setupitaliccorrection[global, always]

% Em => Kursiv (statt slanted)
\setupbodyfontenvironment[default][em=italic]

% Keine hängende Punktuation links:
\setupfontprotrusion [quality] [left=0,right=0.5]

% Defaults
\definefontfeature[default][default][
	script=latn,
	protrusion=quality,
	expansion=quality,
	itlc=yes,
	textitalics=yes,
	onum=yes,
	pnum=yes,
	blockligatures=yes,
	compose=yes,
	]

% Smallcaps
\definefontfeature[smallcaps][
	script=latn,
	protrusion=quality,
	expansion=quality,
	smcp=yes,
	onum=yes,
	pnum=yes,
	compose=yes,
	]

% Superior-Figures (notwendig bei Cardo), sonst superiors schon so.
\definefontfeature[f:superiors][onum=no,pnum=no,sups=yes]

% Oldstyle-Numbers feature (wichtig,
\definefontfeature[f:oldstylenumbers][onum=yes]

% Schriftarten

\definefontfamily[hebrew][rm][SBL Hebrew][features=hebrew]

\definefallbackfamily[mainface][rm][SBL Greek]  [preset=range:greek, force=yes]
\definefallbackfamily[mainface][rm][SBL Hebrew] [range=hebrew, force=yes, features=hebrew]
%\definefallbackfamily[mainface][rm][Amiri] [range=arabic, force=yes, features=arabic,scale=.8]
\definefallbackfamily[mainface][rm][Scheherazade] [range=arabic, force=yes, features=arabic, scale=1.2]

\definefallbackfamily[mainface][rm][Gentium] [range={cyrillic,cyrillicextendeda,cyrillicextendedb,cyrillicsupplement}, force=yes]

\definefallbackfamily[mainface][rm][Segoe UI Symbol][range=0x00000-0xFFFFF, check=yes,force=no,scale=0.85]

% --- DHR-Override: Roman + Sans gewechselt ---
\definefontfamily[mainface][rm][TeX Gyre Pagella]
\definefontfamily[mainface][mm][Latin Modern Math]
\definefontfamily[mainface][ss][TeX Gyre Heros]
\definefontfamily[mainface][tt][DejaVu Sans Mono][features=none,scale=0.8]

\setupbodyfont[
    mainface,
    11pt,
    ]
