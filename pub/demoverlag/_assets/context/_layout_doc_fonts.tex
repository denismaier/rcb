\startenvironment *

% Demo-Font-Setup (Demoverlag-Ebene): nur ConTeXt-gebindete Fonts,
% keine lokal installierten Schriften, keine Skript-Fallbacks
% (Hebräisch etc. ist kein Demo-Content — die produktive Baseline
% mit SBL/Scheherazade/Gentium-Fallbacks steht in
% pub/_assets/context/_layout_doc_fonts.tex).
% Journal-Level-Shadowing: diese Datei ersetzt die Baseline fuer
% alle Artikel unter demoverlag/. Default-Style: rm (TeX Gyre
% Pagella); dhr schaltet per eigener Kopie auf ss (TeX Gyre Heros).

% Kursivkorrektur
\setupitaliccorrection[global, always]

% Em => Kursiv (statt slanted)
\setupbodyfontenvironment[default][em=italic]

% Keine hängende Punktuation links:
\setupfontprotrusion [quality] [left=0,right=0.5]

% Defaults
\definefontfeature[default][default][
	script=latn,
	protrusion=quality,
	expansion=quality,
	itlc=yes,
	textitalics=yes,
	onum=yes,
	pnum=yes,
	blockligatures=yes,
	compose=yes,
	]

% Smallcaps
\definefontfeature[smallcaps][
	script=latn,
	protrusion=quality,
	expansion=quality,
	smcp=yes,
	onum=yes,
	pnum=yes,
	compose=yes,
	]

% Superior-Figures
\definefontfeature[f:superiors][onum=no,pnum=no,sups=yes]

% Oldstyle-Numbers feature
\definefontfeature[f:oldstylenumbers][onum=yes]

% Schriftarten
\definefontfamily[mainface][rm][TeX Gyre Pagella]
\definefontfamily[mainface][mm][Latin Modern Math]
\definefontfamily[mainface][ss][TeX Gyre Heros]
\definefontfamily[mainface][tt][TeX Gyre Cursor][features=none]

\setupbodyfont[
    mainface,
    rm,
    11pt,
    ]

% Symbole
\usesymbols[cc]

\stopenvironment
