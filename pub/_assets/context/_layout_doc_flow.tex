% Autoren
\definenamespace
   [documentauthor] % name of internal variables
   [type=module,
    name=author, 
    command=yes, % Create \defineauthor
    style=yes, % Create \useauthorstyleandcolor 
    setup=list, % Create \setupauthor
    parent=documentauthor,
  ]

\def\AuthorList{}

\define[1]\useauthor
    {\edef\currentauthor{#1}%
	{\doifemptyelse  {\authorparameter{surname}}   
		{
			{\useauthorstyleandcolor{style}{color}
			\authorparameter{name}
			}%
		} 
		{
			{\useauthorstyleandcolor{style}{color}
			\authorparameter{given-names}
			\authorparameter{surname}}%}
			\doiftext{\authorparameter{role}}
			{\space(\authorparameter{role})}
		}
	}
     \par
	 \blank[small]
     \authorparameter{affiliation}
	 \par
     \mono{\authorparameter{email}}
     \blank
     }
	 
\define[1]\useauthorwithoutmail
    {\edef\currentauthor{#1}%
	{\doifemptyelse  {\authorparameter{surname}}   
		{
			{\useauthorstyleandcolor{style}{color}
			\authorparameter{name}
			}%
		} 
		{
			{\useauthorstyleandcolor{style}{color}
			\authorparameter{given-names}
			\authorparameter{surname}}%}
		}
	}
     \par
	 \blank[small]
     \authorparameter{affiliation}
	 \par
     }

\define[1]\useauthorwithcommas
    {\edef\currentauthor{#1}%
	{\doifemptyelse  {\authorparameter{surname}}   
		{
			{\useauthorstyleandcolor{style}{color}
			\authorparameter{name}
			}%
		} 
		{
			{\useauthorstyleandcolor{style}{color}
			\authorparameter{given-names}
			\authorparameter{surname}}%}
		}
	}
    }

% Subtitle Rendering
\startsetups placesubtitle
	\doifnot{\documentvariable{subtitle}} {}
	  {\blank[small]
	   {\doifmodeelse{article}{\tfa}{\tf}\setupinterlinespace
	    \documentvariable{subtitle}}\par}
\stopsetups

% Autoren platzieren
\startsetups placeauthors
  	%\setupnotation[footnote][numberconversion=set 2]
	\blank
	\startalignment[flushleft]
	  \setupindenting[none]
	  \setupwhitespace[none]
	  \processcommacommand[\AuthorList]\useauthor  
	\stopalignment
	%\setupnotation[footnote][numberconversion=set 1]
	\def\AuthorList{}
	\blank[big]
	\noindentation
\stopsetups


% Abstract platzieren
\startsetups abstract
	\doifnot{\documentvariable{abstract}} {}
	{  
	  %\startabstractbackground
	  \startabstract
		\documentvariable{abstract}
	  \stopabstract
	  %\stopabstractbackground
	}
\stopsetups

\startsetups journalinfostitle
	\startalignment[right]
	  \setupindenting[none]
	  \setupwhitespace[none]
	  \switchtobodyfont[9pt]
		{\sc Judaica: Neue digitale Folge}
		\nobreakspace
		\documentvariable{volume}
		\doifnot{\documentvariable{issue}} {} 
			{, \labeltext{nr}\narrownobreakspace\documentvariable{issue}}
		\space
		(\documentvariable{pub-year})
		\doifnot{\documentvariable{issue-title}} {} 
				{
				  \par
				  Special Issue:\space
				  \emph{\documentvariable{issue-title}}
				  \par
				  % TODO -> Ed. / Hrsg. einbauen
				  \labeltext{hg}
				}
		\par
		{\feature[-]{f:oldstylenumbers}\goto{\documentvariable{doi}}[url(\documentvariable{doi})]
		}
		\par
		\symbol[cc][cc] \symbol[cc][by] -- {\sc cc by} 4.0
	\stopalignment
	\blank[small]
	\hrule
	\blank[big]
\stopsetups

\startsetups author-comment
	\doifnot{\documentvariable{notes}} {} 
	  {\footnotetext[+]{\documentvariable{notes}}\resetnumber[footnote]\noindentation}
\stopsetups
