	patterns={de,agr,he},
	goodies={lang-de.llg},
	lefthyphenmin=2,
	righthyphenmin=3,
	hyphenmin=6] 
\setuplanguage[en][patterns={en,agr,he},goodies={lang-en.llg},spacing=packed] 
\setuplanguage[fr][patterns={fr,agr,he}]

\mainlanguage[de]
